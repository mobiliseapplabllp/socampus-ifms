setup for seding push notification in ios device

Step 1. Createing a project in firebase console and download GoogleService-info.plist file

Step 2. go your identifier and tick Push Notification(NOTES: NO NEED TO ATTACH CERTIFICATE)

Step 3. go to keys if already is created then download .p8 file and also APNS ENVIRONMENT is Sandbox & Production (NOTE agar yeah download ho chuki hai toh frr download nhi hogi ishe safe jagah rakhe )

Step 4. Go to firebase console -> project -> cloud messaging -> select App
   => APNs Authentication Key(here uploadyour .p8) with key ID and Team ID
    -> kEY_ID GO TO YOUR KEYS AND FIND KEY_ID 
    -> TEAM_ID go to https://developer.apple.com/account and go to member details fine Team ID here

step 5. open xcode this command npx cap open ios and Drag a GoogleService-Info.plist file in App/App/GoogleService-info.plist here

Step 6. open ios/App/Podfile
  target 'App' do
  capacitor_pods
  # Add your Pods here
  pod 'Firebase/Messaging' // ADD THIS LINE
end

STEP 7 RUN COMMAND 
 => npx cap update ios

    func application(_ application: UIApplication, didRegisterForRemoteNotificationsWithDeviceToken deviceToken: Data) {
        print("📦 APNs token received: \(deviceToken.map { String(format: "%02.2hhx", $0) }.joined())")    
        // Send APNs token to Firebase
        Messaging.messaging().apnsToken = deviceToken

        // Request the FCM token from Firebase
        Messaging.messaging().token { token, error in
            if let error = error {
                print("❌ Error getting FCM token: \(error.localizedDescription)")
                NotificationCenter.default.post(name: Notification.Name(CAPNotifications.DidFailToRegisterForRemoteNotificationsWithError.name()), object: error)
            } else if let token = token {
                print("✅ FCM Token from Firebase: \(token)")
                NotificationCenter.default.post(name: Notification.Name(CAPNotifications.DidRegisterForRemoteNotificationsWithDeviceToken.name()), object: token)
            }
        }
    }

      NotificationCenter.default.post(name: .capacitorDidFailToRegisterForRemoteNotifications, object: error)
    }
